二维码合成海报

发表于2026-04-14|更新于2026-08-17|功能
|浏览量:

功能：根据分享码和项目名，生成二维码并合成到海报模版中。

方案：使用插件qrcode 生成二维码，使用插件canvas-to-image 合成海报。

async getQRCode(url) {
  try {
    let opts = {
      errorCorrectionLevel: 'L', // 容错级别
      type: 'image/png', // 二维码类型
      margin: 0,
      width: 480, // 二维码宽度
      height: 480, // 二维码高度
      color: {
        dark: '#333', // 前景色
        light: '#fff' // 背景色
      }
    }
    const qrCodeCanvas = document.getElementById('qrCodeCanvas')
    await QRCode.toCanvas(qrCodeCanvas, url, opts, (error) => {
      console.error(error); // 生成二维码异常
    })
    this.getImageUrl(qrCodeCanvas.toDataURL()); // 将当前元素内容编码成一张图片的 Data URI
  } catch (error) {
    console.log(error);
  }
},
// 二维码图片和背景图合成海报
createPoster(qrcodeUrl) {
  try {
    const canvas = document.getElementById('posterCanvas');
    const bacImg = new Image();
    bacImg.src = this.backgroundImg; // 设置海报背景图
    let that = this;
    bacImg.onload = function () {
      canvas.width = bacImg.width; // 背景图宽
      canvas.height = bacImg.height; // 背景图高
      let context = canvas.getContext('2d');
      context.rect(0, 0, canvas.width, canvas.height);
      context.fillStyle = '#fff';
      context.fill();
      context.drawImage(bacImg, 0, 0, canvas.width, canvas.height); // 绘制背景图
      const qrCodeImg = new Image();
      qrCodeImg.src = qrcodeUrl; // 设置二维码图
      qrCodeImg.onload = function () {
        let x = 0; // 二维码以背景图横向偏离位置
        let y = 0; // 二维码以背景图纵向偏离位置
        context.drawImage(qrCodeImg, x, y, qrCodeImg.width, qrCodeImg.height); // 绘制二维码图
        that.placardUrl = canvas.toDataURL("image/jpeg", 1.0); // 合成海报图片base64编码
      }
    }
  } catch (error) {
    console.log(error);
  }
}
